\begin{frame}{Преимущества и ограничения}
\begin{columns}[T,onlytextwidth]
\begin{column}{.50\textwidth}
\textcolor{courseaccent}{\textbf{Преимущества}}
\begin{itemize}
\item Физическая и геометрическая интерпретируемость элементов.
\item Предварительное планирование миссий: оценка основных характеристик орбиты.
\item Стандарты и совместимость: использование элементов в каталогах и расчётных средствах для обмена орбитальными данными.
\end{itemize}
\end{column}
\begin{column}{.46\textwidth}
\textcolor{courseaccent}{\textbf{Ограничения}}
\begin{itemize}
\item При $e=0$ аргумент перицентра $\omega$ и истинная аномалия $\nu$ по отдельности не определены.
\item При $i=0$ или $\pi$ углы $\Omega$ и $\omega$ по отдельности не определены.
\item Вблизи вырожденных случаев преобразование в угловые элементы плохо обусловлено: возрастает чувствительность к погрешностям исходных данных.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Декартовы элементы}
\begin{block}{Определение}
\justifying
Положение и скорость КА задаются шестью компонентами в выбранной системе координат на эпоху $t_0$.
\end{block}
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\textcolor{courseaccent}{\textbf{Вектор состояния}}
\par\smallskip
$\mathbf y=(x,y,z,v_x,v_y,v_z)^{\mathsf T}.$
\par\medskip
$\mathbf r=(x,y,z)^{\mathsf T}$ --- радиус-вектор.\par
$\mathbf v=(v_x,v_y,v_z)^{\mathsf T}$ --- скорость.
\par\medskip
\textbf{Единицы:}\par
$x,y,z$ --- м или км;\par
$v_x,v_y,v_z$ --- м/с или км/с.
\end{column}
\begin{column}{.48\textwidth}
\textcolor{courseaccent}{\textbf{Отсчёт и свойства}}
\begin{itemize}
\item Компоненты --- знаковые проекции векторов $\mathbf r,\mathbf v$ на оси СК.
\item В геоцентрической СК начало совпадает с центром масс Земли.
\item Нет угловых вырождений при $e=0$ и $i=0,\pi$.
\end{itemize}
\end{column}
\end{columns}
\end{frame}

% Набор (a,h,k,p,q,lambda) перенесён из прошлогодней лекции.
% Он использует среднюю долготу; стандартный MEE использует фокальный
% параметр вместо a и истинную долготу вместо lambda.
% https://orekit.org/site-orekit-development/apidocs/org/orekit/files/ccsds/ndm/odm/ocm/OrbitElementsType.html
\begingroup
\setbeamerfont{frametitle}{size*={17.5pt}{23pt}}
\begin{frame}{Модифицированные равноденственные элементы}
\begin{columns}[T,onlytextwidth]
\begin{column}{.49\textwidth}
\begin{block}{Набор $(a,h,k,p,q,\lambda)$}
\centering
$\displaystyle
\begin{gathered}
h = e\sin(\omega+I\Omega),\quad
k = e\cos(\omega+I\Omega),\\[4pt]
p = \left[\tan\frac{i}{2}\right]^I\sin\Omega,\quad
q = \left[\tan\frac{i}{2}\right]^I\cos\Omega,\\[4pt]
\lambda = M+\omega+I\Omega.
\end{gathered}
$\par
\end{block}
$a$ --- большая полуось, м или км;\par
$h,k,p,q$ --- безразмерные;\par
$\lambda$ --- средняя долгота, рад.
\end{column}
\begin{column}{.47\textwidth}
\begin{block}{Ретроградный фактор}
\centering
$\displaystyle
I=\begin{cases}
+1,&0\le i\le\pi/2,\\
-1,&\pi/2<i\le\pi.
\end{cases}
$\par
\end{block}
\textbf{Особенности каждого набора:}\par\smallskip
При $I=+1$: $\displaystyle\lim_{i\to\pi}\sqrt{p^2+q^2}=\infty.$\par\smallskip
При $I=-1$: $\displaystyle\lim_{i\to0}\sqrt{p^2+q^2}=\infty.$\par\smallskip
При указанном выборе $I$ набор регулярен для круговых и экваториальных орбит.
\end{column}
\end{columns}
\end{frame}
\endgroup

% https://www.orekit.org/site-orekit-development/apidocs/org/orekit/frames/LOFType.html
\begin{frame}{Орбитальная система координат}
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}
\centering
\input{images/orbital-frame.tex}
\par\smallskip
{\usebeamerfont{caption}\color{coursemuted}
$\mathbf R$ --- радиус-вектор от центра Земли до центра масс КА.\par}
\smallskip
Правый базис: $\mathbf e_r\times\mathbf e_t=\mathbf e_z$.
\end{column}
\begin{column}{.49\textwidth}
\begin{block}{Начало координат}
$O$ --- центр масс КА.
\end{block}
\textbf{Радиальная ось $\mathbf e_r$.}\par
Сонаправлена радиус-вектору $\mathbf R$.
\par
\textbf{Трансверсальная ось $\mathbf e_t$.}\par
{\justifying
Лежит в плоскости орбиты, перпендикулярна $\mathbf e_r$ и направлена в сторону движения КА.\par}
\par
\textbf{Нормальная ось $\mathbf e_z$.}\par
{\justifying
Перпендикулярна плоскости орбиты и дополняет оси до правой тройки.\par}
\end{column}
\end{columns}

\begin{frame}{Задача маневрирования}
\begin{block}{Определение}

Маневрирование --- целенаправленное изменение параметров орбиты космического аппарата с помощью двигательной установки.
\end{block}

\justifying

\end{block}
\textbf{Параметры манёвров:} моменты выполнения, направления и величины импульсов скорости либо характеристики включений ДУ.

При оптимизации дополнительно задаётся \textbf{критерий качества}.
\end{frame}

\begin{frame}{Силы, действующие на КА}
Геоцентрическая СК с невращающимися осями:
\par\smallskip
\centerline{$\displaystyle\ddot{\mathbf r}=\mathbf a_0+\mathbf a_{\text{возм}}+\mathbf a_{\text{ДУ}}.$}
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}
\begin{block}{Притяжение точечного центра}
$\displaystyle\mathbf a_0=-\frac{\mu}{r^3}\mathbf r$,\quad $r=\|\mathbf r\|$.
\par\smallskip
$\mu$ --- гравитационный параметр Земли.
\end{block}
\begin{block}{Управляющее ускорение}
$\displaystyle\mathbf a_{\text{ДУ}}=\frac{\mathbf F_{\text{ДУ}}}{m}$,\quad $m$ --- масса КА.
\end{block}
\end{column}
\begin{column}{.49\textwidth}
\begin{block}{Возмущающие ускорения}
\begin{itemize}
\item Несферичность поля Земли.
\item Притяжение Луны и Солнца.
\item Сопротивление атмосферы.
\item Давление солнечного излучения.
\end{itemize}
\end{block}
\end{column}
\end{columns}
Состав учитываемых воздействий зависит от орбиты, свойств КА, длительности прогноза и требуемой точности.
\end{frame}

\begin{frame}{Расчётная модель баллистики}
\begin{columns}[T,onlytextwidth]
\begin{column}{.30\textwidth}
\begin{block}{Центральное поле}
Только притяжение точечного центра.\par\smallskip
Кеплерово движение.
\end{block}
\end{column}
\begin{column}{.31\textwidth}
\begin{block}{С возмущениями}
Дополнительно, например, $J_2$ и сопротивление атмосферы.
\end{block}
\end{column}
\begin{column}{.35\textwidth}
\begin{block}{Линеаризованная}
Приближение по малым отклонениям от опорного движения.
\end{block}
\end{column}
\end{columns}

\begin{frame}{Модель космического аппарата}
Характеристики КА определяют допустимые управляющие воздействия.
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Масса и ресурсы}
Масса $m(t)$ и запас рабочего тела.\par
Изменение массы при работе ДУ.
\end{block}
\begin{block}{Ориентация}
Допустимые направления тяги.\par
Время разворота и точность наведения.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{Двигательная установка}
Тяга, удельный импульс, режимы.\par
Расположение двигателей.
\end{block}
\begin{block}{Исполнение манёвра}
Задержки срабатывания ДУ.\par
Ошибки тяги, времени и направления.
\end{block}
\end{column}
\end{columns}
\smallskip
Масса, форма и ориентация КА также влияют на аэродинамические силы и давление излучения.
\end{frame}

\begin{frame}{Условия задачи и критерий качества}
Модель определяет возможное движение; условия задачи --- требуемый результат.
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Начальные условия}
Эпоха и начальное состояние КА.\par
Состояния целей, доступные ресурсы и погрешности исходных данных.
\end{block}
\begin{block}{Ограничения}
Безопасность, допустимые интервалы манёвров, технические и полётные условия.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{Конечные условия}
Требуемое состояние в конечный момент или допустимая область на интервале движения.
\end{block}
\begin{block}{Критерий качества}
Выбирает предпочтительное решение.\par
Например, для импульсной модели:\par
$\displaystyle J=\sum_i\|\Delta\mathbf v_i\|\to\min$.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Переход}

\begin{columns}[T,onlytextwidth]

\justifying
Начальная орбита и начальная эпоха.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{Конечные условия}
\justifying
Конечная орбита и конечная эпоха. Конечная эпоха может быть задана или оставаться свободной.
\end{block}
\end{column}
\end{columns}
\begin{block}{Возможные ограничения}
\begin{itemize}
\item Допустимые интервалы включений ДУ.
\item Минимальная высота орбиты и доступный запас рабочего тела.
\end{itemize}
\end{block}
\textbf{Критерий:} например, минимум расхода рабочего тела.
\end{frame}

Начальное состояние КА и начальная эпоха.

\end{column}
\begin{column}{.48\textwidth}
\begin{block}{Конечные условия}

Конечное состояние КА и конечная эпоха.

\end{column}
\end{columns}
\begin{block}{Возможные ограничения}
\begin{itemize}

\item Безопасность промежуточного движения и ресурсы КА.

\begin{frame}{Уклонение}
Изменение движения КА для предотвращения опасного сближения.
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Начальные условия}
\justifying
Состояния объектов и их ковариации; начальная эпоха.
\end{block}
\end{column}
\begin{column}{.48\textwidth}

Допустимый риск столкновения на заданном интервале.

\end{column}
\end{columns}
\begin{block}{Возможные ограничения}
\begin{itemize}

\end{itemize}
\end{block}
\textbf{Критерий:} например, минимум расхода или отклонения от рабочей орбиты.

\begin{frame}{Формирование конфигурации}
Размещение КА спутниковой системы по заданным орбитальным позициям.
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Начальные условия}
\justifying
Состояния всех КА после выведения или разведения; доступные ресурсы каждого аппарата.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{Конечные условия}
\justifying
Заданные орбиты и фазовые соотношения КА. Например, требуемые разности $\Omega$ и средней долготы $\lambda$.
\end{block}
\end{column}
\end{columns}
\begin{block}{Возможные ограничения}
\begin{itemize}
\item Срок формирования и индивидуальные ресурсы КА.
\item Безопасные расстояния и доступные интервалы манёвров.
\end{itemize}
\end{block}
\textbf{Критерий:} например, минимум суммарного расхода рабочего тела.
\end{frame}

\begin{frame}{Поддержание конфигурации}
Сохранение требуемой конфигурации спутниковой системы при возмущениях.
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Начальные условия}
\justifying
Фактические состояния КА, отклонения от номинальной конфигурации и прогноз их изменения.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{Конечные условия}

\end{block}
\end{column}
\end{columns}
\begin{block}{Возможные ограничения}
\begin{itemize}
\item Частота коррекций, интервалы включений и ресурсы каждого КА.
\item Безопасные расстояния и требования основной миссии.
\end{itemize}
\end{block}
\textbf{Критерий:} например, минимум расхода за интервал поддержания.
\end{frame}
