% Orthographic views in component space (s_r,s_t,s_z).
% Common camera: azimuth 50 degrees, elevation 25 degrees.
% A unit segment is 1.55 cm before projection. All curves below
% are constructed in 3D; only TikZ's coordinate basis projects them.
\begin{tikzpicture}[>=Latex,line cap=round,line join=round]
\pgfmathsetmacro{\hodNdcSquared}{(1.07+sqrt(.5049))/2}
\pgfmathsetmacro{\hodNdcNormal}{sqrt(\hodNdcSquared-.04)}
\pgfmathsetmacro{\hodNdcContactCos}{.12/(\hodNdcSquared-.16)}
\pgfmathsetmacro{\hodNdcContactSin}{sqrt(1-\hodNdcContactCos^2)}
\foreach \hodPanel/\hodTitle in {0/{Узловой (NC)},1/{Смещённый (NDC)},2/{Особый (SC)}}{
 \begin{scope}[shift={({5.45*\hodPanel},0)}]
  % Silhouette of the transparent unit sphere.
  \fill[figplane!45!white] (0,0) circle[radius=1.55cm];
  \begin{scope}[
   x={(-1.18736889cm,-0.42106336cm)},
   y={(0.99632080cm,-0.50180377cm)},
   z={(0cm,1.40477707cm)}]
   % The plane contains the s_t axis and a unit vector in (s_r,s_z).
   % These are drawing coordinates only, not notation in the text.
   \ifcase\hodPanel
    % lambda_1=lambda_3=lambda_4=0, lambda_2=.275,
    % lambda_5=sqrt(1-.275^2).
    \pgfmathsetmacro{\hodPlaneRadial}{-.275}
    \pgfmathsetmacro{\hodPlaneNormal}{sqrt(1-.275^2)}
   \or
    % lambda_1=.15, lambda_2=.2, lambda_3=lambda_4=0,
    % lambda_5=sqrt(hodNdcSquared-.04).
    \pgfmathsetmacro{\hodPlaneRadial}{-.2/sqrt(\hodNdcSquared)}
    \pgfmathsetmacro{\hodPlaneNormal}{\hodNdcNormal/sqrt(\hodNdcSquared)}
   \or
    % lambda_1=lambda_3=lambda_4=0, lambda_2=.5, lambda_5=sqrt(3)/2.
    \pgfmathsetmacro{\hodPlaneRadial}{-.5}
    \pgfmathsetmacro{\hodPlaneNormal}{sqrt(3)/2}
   \fi
   % Back halves of three great circles.
   \draw[figgray!45,thin,densely dashed]
    plot[domain=0:360,samples=101,variable=\hodAngle]
        ({cos(\hodAngle)},{sin(\hodAngle)},0);
   \draw[figgray!45,thin,densely dashed]
    plot[domain=0:360,samples=101,variable=\hodAngle]
        ({cos(\hodAngle)},0,{sin(\hodAngle)});
   \draw[figgray!45,thin,densely dashed]
    plot[domain=0:360,samples=101,variable=\hodAngle]
        (0,{cos(\hodAngle)},{sin(\hodAngle)});
   % A translucent patch of the hodograph plane through the origin.
   \filldraw[fill=accent,fill opacity=.09,draw=accent!35,line width=.4pt]
    ({-1.08*\hodPlaneRadial},-1.08,{-1.08*\hodPlaneNormal})--
    ({-1.08*\hodPlaneRadial},1.08,{-1.08*\hodPlaneNormal})--
    ({1.08*\hodPlaneRadial},1.08,{1.08*\hodPlaneNormal})--
    ({1.08*\hodPlaneRadial},-1.08,{1.08*\hodPlaneNormal})--cycle;
   % Visible halves of the same great circles: dot(view,point) >= 0.
   \draw[figgray!60,thin]
    plot[domain=-40:140,samples=61,variable=\hodAngle]
        ({cos(\hodAngle)},{sin(\hodAngle)},0);
   \pgfmathsetmacro{\hodFrontXZ}{atan2(sin(25),cos(25)*cos(50))}
   \draw[figgray!60,thin]
    plot[domain={\hodFrontXZ-90}:{\hodFrontXZ+90},samples=61,variable=\hodAngle]
        ({cos(\hodAngle)},0,{sin(\hodAngle)});
   \pgfmathsetmacro{\hodFrontYZ}{atan2(sin(25),cos(25)*sin(50))}
   \draw[figgray!60,thin]
    plot[domain={\hodFrontYZ-90}:{\hodFrontYZ+90},samples=61,variable=\hodAngle]
        (0,{cos(\hodAngle)},{sin(\hodAngle)});
   % Signed coordinate axes in 3D.
   \draw[figgray!55,densely dashed] (-1.18,0,0)--(0,0,0);
   \draw[figgray!55,densely dashed] (0,-1.18,0)--(0,0,0);
   \draw[figgray!55,densely dashed] (0,0,-1.18)--(0,0,0);
   \draw[->,figink!75] (0,0,0)--(1.34,0,0)
    node[below left,inner sep=1.5pt] {$s_r$};
   \draw[->,figink!75] (0,0,0)--(0,1.43,0)
    node[below right,inner sep=1.5pt] {$s_t$};
   \draw[->,figink!75] (0,0,0)--(0,0,1.34)
    node[above,inner sep=1.5pt] {$s_z$};
   \ifcase\hodPanel
    \draw[accent,line width=1.3pt]
     plot[domain=0:360,samples=161,variable=\hodAngle]
      ({-.275*cos(\hodAngle)},{.55*sin(\hodAngle)},
       {sqrt(1-.275^2)*cos(\hodAngle)});
    \foreach \hodSign in {-1,1}{
     \draw[->,accent,line width=.6pt] (0,0,0)--
      ({-.275*\hodSign},0,{sqrt(1-.275^2)*\hodSign});
     \filldraw[fill=fignormal,draw=white,line width=.4pt]
      ({-.275*\hodSign},0,{sqrt(1-.275^2)*\hodSign}) circle[radius=1.9pt];}
   \or
    \draw[accent,line width=1.3pt]
     plot[domain=0:360,samples=161,variable=\hodAngle]
      ({-.2*cos(\hodAngle)},{.3+.4*sin(\hodAngle)},
       {\hodNdcNormal*cos(\hodAngle)});
    \foreach \hodSign in {-1,1}{
     \draw[->,accent,line width=.6pt] (0,0,0)--
      ({-.2*\hodNdcContactSin*\hodSign},{.3+.4*\hodNdcContactCos},
       {\hodNdcNormal*\hodNdcContactSin*\hodSign});
     \filldraw[fill=fignormal,draw=white,line width=.4pt]
      ({-.2*\hodNdcContactSin*\hodSign},{.3+.4*\hodNdcContactCos},
       {\hodNdcNormal*\hodNdcContactSin*\hodSign}) circle[radius=1.9pt];}
    \node[accent,inner sep=0pt] at (0,.3,0) {$\times$};
   \or
    \draw[accent,line width=1.3pt]
     plot[domain=0:360,samples=161,variable=\hodAngle]
      ({-.5*cos(\hodAngle)},{sin(\hodAngle)},
       {sqrt(3)/2*cos(\hodAngle)});
    \foreach \hodAngle in {20,145,255}{
     \draw[->,accent,line width=.6pt] (0,0,0)--
      ({-.5*cos(\hodAngle)},{sin(\hodAngle)},
       {sqrt(3)/2*cos(\hodAngle)});
     \filldraw[fill=fignormal,draw=white,line width=.4pt]
      ({-.5*cos(\hodAngle)},{sin(\hodAngle)},
       {sqrt(3)/2*cos(\hodAngle)}) circle[radius=1.9pt];}
   \fi
   \fill[figink] (0,0,0) circle[radius=.8pt];
   \node[below left,inner sep=2pt] at (0,0,0) {$0$};
  \end{scope}
  \draw[figgray!65,line width=.45pt] (0,0) circle[radius=1.55cm];
  \node[align=center] at (0,-2.40) {\hodTitle};
 \end{scope}
}
\end{tikzpicture}
